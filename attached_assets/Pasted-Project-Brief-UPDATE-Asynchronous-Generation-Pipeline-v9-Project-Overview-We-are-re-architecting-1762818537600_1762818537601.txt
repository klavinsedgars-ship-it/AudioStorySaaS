Project Brief UPDATE: Asynchronous Generation Pipeline (v9)
Project Overview: We are re-architecting the story creation flow to fix long wait times and server timeouts. When a user clicks "Create Audio & Save," the server will immediately save the story and deduct the credit, then perform the slow audio generation and illustration generation in the background.

The user will be redirected to a new "magical loading" page that shows the progress of their story being created.

Core Goals:

Make the /api/generate-story-audio route respond instantly (in < 1 second).

Create two new background (async) jobs on the server: one for audio, one for images.

Add a status field to the stories table to track progress.

Create a new "magical" loading page (/story/generating/:storyId) that polls for this status.

Implement the asynchronous illustration generation as planned in the previous prompt.

// === [NEW] BACKGROUND JOB 2: GENERATE ILLUSTRATION ===
async function generateAndSaveIllustration(storyId: string, storyText: string, heroName: string) {
  try {
    // Wait until audio is done before starting
    let story = await storage.getStory(storyId);
    while (story && story.status !== 'gen_image') {
      if (story.status === 'failed_audio' || story.status === 'failed_image') return;
      await new Promise(resolve => setTimeout(resolve, 2000)); // Wait 2s and check again
      story = await storage.getStory(storyId);
    }
    
    // 1. Summarize Text for Image Prompt (Fast)
    const summarizerPrompt = `
      You are an AI assistant for a children's storybook app. 
      Summarize the following story text into a single, visually descriptive DALL-E 3 prompt.
      Style: "A whimsical, magical, children's book illustration"
      Example: "A whimsical, magical, children's book illustration of a young boy named Leo and his pet dog flying in a rocket ship past a smiling, ringed planet."
      STORY TEXT:
      ${storyText.substring(0, 1500)}...
    `;
    const summaryResponse = await openai.chat.completions.create({
      model: "gpt-4o-mini",
      messages: [{ role: "user", content: summarizerPrompt }], temperature: 0.2,
    });
    const imagePrompt = summaryResponse.choices[0].message.content;

    // 2. Generate Image with DALL-E 3 (Long wait)
    const imageResponse = await openai.images.generate({
      model: "dall-e-3", prompt: imagePrompt!, n: 1, size: "1024x1024",
    });
    const imageUrl = imageResponse.data[0].url; // This is a temp URL

    // [MODIFIED] We must download this image and save it to our *own* storage, as the OpenAI URL expires.
    const imageFetch = await fetch(imageUrl!);
    const imageBuffer = await imageFetch.arrayBuffer();
    const imageFilename = `image-${storyId}-${Date.now()}.png`;
    const finalImagePath = await uploadAudioToStorage(imageBuffer, imageFilename); // Re-using this uploader
    const finalPublicUrl = `/api/audio/${storyId}?type=image`; // We will serve it via our own streaming endpoint
    
    // 3. Update DB
    await storage.updateStoryImage(storyId, finalPublicUrl); // Save our internal path
    await storage.updateStoryStatus(storyId, 'complete'); // All done!
    console.log(`[IMAGE] Successfully generated image for story ${storyId}`);

  } catch (error) {
    console.error(`[IMAGE] Failed to generate image for story ${storyId}:`, error);
    await storage.updateStoryStatus(storyId, 'failed_image');
  }
}

// client/src/pages/story-loading.tsx
import { useQuery } from "@tanstack/react-query";
import { useRoute, useLocation } from "wouter";
import { Loader2, Wand2, CheckCircle2, Music, Image } from "lucide-react";
import { Button } from "@/components/ui/button";

// A hook to fetch the story status, polling every 3 seconds
function useStoryStatus(storyId: string) {
  const [, setLocation] = useLocation();

  return useQuery({
    queryKey: ["/api/story/status", storyId],
    queryFn: async () => {
      const res = await fetch(`/api/story/status/${storyId}`, { credentials: "include" });
      if (!res.ok) throw new Error("Could not fetch story status");
      const data = await res.json();

      // [NEW] When complete, auto-redirect to bookshelf
      if (data.status === 'complete') {
        setTimeout(() => {
          setLocation("/bookshelf");
        }, 1500); // Wait 1.5s on "complete" so user can read it
      }
      return data;
    },
    refetchInterval: 3000, // Poll every 3 seconds
  });
}

// Helper for loading messages
const getLoadingMessage = (status: string) => {
  switch (status) {
    case 'pending':
    case 'gen_audio':
      return "Warming up the narrator's voice...";
    case 'gen_image':
      return "Painting the cover illustration...";
    case 'complete':
      return "Your story is ready!";
    default:
      return "Your story is in the magical oven...";
  }
};

export default function StoryLoading() {
  const [, params] = useRoute("/story/generating/:storyId");
  const storyId = params?.storyId || "";
  
  const { data: story, isLoading }_ = useStoryStatus(storyId);
  const status = story?.status || 'pending';

  return (
    <div className="min-h-screen bg-gradient-to-b from-purple-50 via-background to-background dark:from-purple-950/20 dark:via-background dark:to-background pt-24 pb-16 flex items-center justify-center">
      <div className="max-w-md mx-auto text-center p-8">
        <h1 className="font-display text-4xl font-bold mb-6 bg-gradient-to-r from-primary to-purple-600 bg-clip-text text-transparent">
          Adding the Magic...
        </h1>
        <p className="text-xl text-muted-foreground mb-12">
          {getLoadingMessage(status)}
        </p>

        <div className="space-y-6">
          {/* Step 1: Text */}
          <div className="flex items-center gap-4 p-4 bg-card border rounded-lg shadow-sm">
            <CheckCircle2 className="w-6 h-6 text-green-500" />
            <span className="font-medium text-lg">Story text approved</span>
          </div>

          {/* Step 2: Audio */}
          <div className="flex items-center gap-4 p-4 bg-card border rounded-lg shadow-sm">
            {status === 'pending' || status === 'gen_audio' ? (
              <Loader2 className="w-6 h-6 text-primary animate-spin" />
            ) : (
              <CheckCircle2 className="w-6 h-6 text-green-500" />
            )}
            <span className="font-medium text-lg">Creating high-quality audio</span>
          </div>

          {/* Step 3: Image */}
          <div className="flex items-center gap-4 p-4 bg-card border rounded-lg shadow-sm">
            {status === 'complete' ? (
              <CheckCircle2 className="w-6 h-6 text-green-500" />
            ) : (
              <Loader2 className={`w-6 h-6 ${status === 'gen_image' ? 'text-primary animate-spin' : 'text-muted-foreground'}`} />
