# Project Brief: Full-Stack Multilingual Audio Story SaaS (v7 - IMPROVED)

## Project Overview
Build a full-stack, monetized, multilingual SaaS application entirely within the Replit ecosystem. The app allows parents to generate personalized audio bedtime stories for their children.

This project uses Replit Auth, Replit Database, and Replit App Storage.

---

## Core Features

- **Replit-Native Stack:** Uses Replit Auth, Replit Database (Key-Value), and Replit App Storage (Object Storage).
- **Monetization:** A credit-based system with one-time purchase packages.
- **Cost-Saving Flow:** A "Text-First Approval" step to prevent wasting credits.
- **Deep Personalization:**
  - Main hero's name
  - Multiple additional names (siblings, friends, pets)
  - Two generation modes: "Choose a Theme" or "Write Your Own Prompt"
- **Advanced i18n:**
  - Geo-IP detection to set the site's default language
  - Full UI translation
  - Global language dropdown in navbar
  - Story text and audio generation in selected language
- **Story Uniqueness:** Backend logic to ensure stories are different each time

---

## Core Tech Stack

- **Framework:** Next.js (with TypeScript)
- **Authentication:** Replit Auth (using useReplit hook)
- **Database:** Replit Database (Key-Value, using @replit/database)
- **File Storage:** Replit App Storage (Object Storage, for MP3s)
- **Styling:** TailwindCSS
- **Animations:** Framer Motion
- **Text Generation:** OpenAI API
- **Audio Generation:** ElevenLabs API
- **Payments:** Stripe (Payment Links & Webhooks)
- **State Management:** React Context (for Language & User)

### 2. Next.js Middleware (middleware.ts)

```typescript
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';
import { LANGUAGE_OPTIONS } from './lib/constants';

export function middleware(request: NextRequest) {
  const existingLang = request.cookies.get('app-lang');
  
  if (!existingLang) {
    const country = request.geo?.country || 'US';
    const matchedLang = LANGUAGE_OPTIONS.find(l => l.country === country);
    const langCode = matchedLang?.code || 'en';
    
    const response = NextResponse.next();
    response.cookies.set('app-lang', langCode, {
      maxAge: 60 * 60 * 24 * 365, // 1 year
      path: '/'
    });
    return response;
  }
  
  return NextResponse.next();
}

export const config = {
  matcher: '/((?!api|_next/static|_next/image|favicon.ico).*)',
};
```

### 3. Global Language Context (context/LanguageContext.tsx)

```typescript
import { createContext, useContext, useState, useEffect, ReactNode } from 'react';
import Cookies from 'js-cookie';

interface LanguageContextType {
  currentLang: string;
  setCurrentLang: (lang: string) => void;
}

const LanguageContext = createContext<LanguageContextType | undefined>(undefined);

export function LanguageProvider({ children }: { children: ReactNode }) {
  const [currentLang, setCurrentLangState] = useState('en');

  useEffect(() => {
    const cookieLang = Cookies.get('app-lang');
    if (cookieLang) {
      setCurrentLangState(cookieLang);
    }
  }, []);

  const setCurrentLang = (lang: string) => {
    setCurrentLangState(lang);
    Cookies.set('app-lang', lang, { expires: 365 });
  };

  return (
    <LanguageContext.Provider value={{ currentLang, setCurrentLang }}>
      {children}
    </LanguageContext.Provider>
  );
}

export function useLanguage() {
  const context = useContext(LanguageContext);
  if (!context) {
    throw new Error('useLanguage must be used within LanguageProvider');
  }
  return context;
}
```

**hooks/useTranslation.ts:**

```typescript
import { useLanguage } from '../context/LanguageContext';
import en from '../locales/en.json';
import lv from '../locales/lv.json';
import es from '../locales/es.json';
import fr from '../locales/fr.json';

const translations: Record<string, any> = { en, lv, es, fr };

export function useTranslation() {
  const { currentLang } = useLanguage();
  
  const t = (key: string): string => {
    const keys = key.split('.');
    let value = translations[currentLang];
    
    for (const k of keys) {
      value = value?.[k];
    }
    
    return value || key;
  };
  
  return { t };
}
```

```typescript
  };

    setIsLoading(true);
    const response = await fetch('/api/generateStoryAudio', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        storyText,
        language: currentLang,
        heroName,
        additionalNames,
        theme: generationType === 'theme' ? selectedTheme : 'Custom',
        customPromptText: generationType === 'custom' ? customPrompt : null

      alert('Story created! Check your bookshelf.');
```

## Phase 4: Backend API Routes (The Engine)

### 1. API: generateStoryText (pages/api/generateStoryText.ts)

```typescript
import { NextApiRequest, NextApiResponse } from 'next';
import { getUser } from '@replit/replit-auth-api';
import { Client } from '@replit/database';
import OpenAI from 'openai';

const db = new Client();
const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });

export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  if (req.method !== 'POST') return res.status(405).end();

  const user = await getUser(req);
  if (!user) return res.status(401).json({ error: 'unauthorized' });

  const { heroName, additionalNames, language, generationType, theme, customPrompt } = req.body;

  // Rate limiting
  const rateLimitKey = `ratelimit_${user.id}_text`;
  const lastRequest = await db.get(rateLimitKey);
  if (lastRequest && Date.now() - lastRequest < 5000) {
    return res.status(429).json({ error: 'too_many_requests', message: 'Please wait 5 seconds between requests.' });
  }
  await db.set(rateLimitKey, Date.now());

  // Check credits (soft check - actual deduction happens on audio generation)
  const profile = await db.get(`profile_${user.id}`);
  if (!profile || profile.credits <= 0) {
    return res.status(402).json({ 
      error: 'insufficient_credits',
      message: 'You need credits to generate stories. Please purchase more.'
    });
  }

  // Generate random plot element for uniqueness
  const randomElements = [
    'a friendly talking moon',
    'a magical sleepy cloud',
    'a tiny owl with stories to tell',
    'a gentle rainbow that hums lullabies',
    'a wise old tree with glowing leaves',
    'a sleepy dragon who collects dreams',
    'a starlight fairy with a soothing voice',
    'a cozy blanket that tells bedtime tales'
  ];

  const randomLocations = [
    'a cozy treehouse',
    'a starlit meadow',
    'a quiet library in the clouds',
    'a warm cave with glowing crystals',
    'a peaceful garden under the moon'
  ];

  const randomChallenges = [
    'helps a lost firefly find its way home',
    'discovers a garden of sleeping flowers',
    'finds the softest pillow in the world',
    'helps the stars prepare for bedtime',
    'learns a magical lullaby from the wind'
  ];

  const plotElement = randomElements[Math.floor(Math.random() * randomElements.length)];
  const location = randomLocations[Math.floor(Math.random() * randomLocations.length)];
  const challenge = randomChallenges[Math.floor(Math.random() * randomChallenges.length)];

  // Build the master prompt
  const taskBlock = generationType === 'theme' 
    ? `**Task: Theme-Based Story**
* **Theme:** ${theme}
* **Your Job:** Create a full story in ${language} based on the characters and this theme.`
    : `**Task: Custom Prompt**
* **User's Request:** ${customPrompt}
* **Your Job:** Use this custom request as the main plot, but you MUST follow all critical rules (soothing tone, sleepy structure, audio-first, and language: ${language}).`;

  const masterPrompt = `You are a master bedtime storyteller. Your purpose is to create a gentle, 5-10 minute audio-ready story. Your tone is soothing, magical, and calm.

**CRITICAL RULES:**
1. **[LANGUAGE]**: You MUST write the entire story in ${language}.
2. **Audio-First:** Use short, clear sentences. Use frequent paragraph breaks for TTS pauses.
3. **Soothing Tone:** NO scary words (monster, terror, scream). The story must be 100% safe.
4. **Uniqueness:** You MUST incorporate these random elements into the story:
   - Plot Element: ${plotElement}
   - Location: ${location}
   - Challenge: ${challenge}
5. **Structure (Required):**
   * **The Yawn:** Start with a sleepy, calm scene.
   * **The Gentle Adventure:** Introduce the plot.
   * **The Sleepy Resolution:** Solve the plot kindly and wind down. The story MUST end with the child feeling safe, warm, and very sleepy.

---

**STORY REQUEST:**

* **Main Hero:** ${heroName}
* **Additional Characters:** ${additionalNames.join(', ')} (Weave these names in as friends/helpers.)
* **Language:** ${language}

---

${taskBlock}

---

Write the complete story now. Remember: soothing, audio-ready, in ${language}, and use the random elements for uniqueness.`;

  try {
    const completion = await openai.chat.completions.create({
      model: 'gpt-4',
      messages: [{ role: 'user', content: masterPrompt }],
      temperature: 0.9, // Higher for more creativity/uniqueness
      max_tokens: 1500
    });

    const storyText = completion.choices[0].message.content;

    return res.status(200).json({ storyText });
  } catch (error) {
    console.error('OpenAI error:', error);
    return res.status(500).json({ 
      error: 'generation_failed',
      message: 'Failed to generate story. Please try again.'
    });
  }
}
```

### 2. API: generateStoryAudio (pages/api/generateStoryAudio.ts)

```typescript
import { NextApiRequest, NextApiResponse } from 'next';
import { getUser } from '@replit/replit-auth-api';
import { Client } from '@replit/database';
import { getLangConfig } from '../../lib/constants';
import { v4 as uuidv4 } from 'uuid';

const db = new Client();

export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  if (req.method !== 'POST') return res.status(405).end();

  const user = await getUser(req);
  if (!user) return res.status(401).json({ error: 'unauthorized' });

  const { storyText, language, heroName, additionalNames, theme, customPromptText } = req.body;

  // STEP 1: Check credits FIRST
  const profile = await db.get(`profile_${user.id}`);
  if (!profile || profile.credits <= 0) {
    return res.status(402).json({ 
      error: 'insufficient_credits',
      message: 'You need credits to create audio. Please purchase more.'
    });
  }

  // STEP 2: Get voice configuration
  const langConfig = getLangConfig(language);
  const voiceId = langConfig.elevenLabsVoiceId;

  // STEP 3: Generate audio with ElevenLabs (can fail safely - no credit loss yet)
  let audioBuffer: Buffer;
  try {
    const response = await fetch(
      `https://api.elevenlabs.io/v1/text-to-speech/${voiceId}`,
      {
        method: 'POST',
        headers: {
          'Accept': 'audio/mpeg',
          'Content-Type': 'application/json',
          'xi-api-key': process.env.ELEVENLABS_API_KEY!
        },
        body: JSON.stringify({
          text: storyText,
          model_id: 'eleven_multilingual_v2',
          voice_settings: {
            stability: 0.5,
            similarity_boost: 0.75
          }
        })
      }
    );

    if (!response.ok) {
      throw new Error(`ElevenLabs API error: ${response.statusText}`);
    }

    audioBuffer = Buffer.from(await response.arrayBuffer());
  } catch (error) {
    console.error('ElevenLabs error:', error);
    return res.status(500).json({ 
      error: 'audio_generation_failed',
      message: 'Failed to generate audio. Please try again. No credits were deducted.'
    });
  }

  // STEP 4: Upload to Replit Storage (can fail safely - no credit loss yet)
  const storyId = uuidv4();
  const timestamp = Date.now();
  const filename = `${user.id}_${timestamp}_${heroName.toLowerCase().replace(/\s/g, '_')}.mp3`;
  
  let publicUrl: string;
  try {
    // Upload to Replit App Storage
    const storageResponse = await fetch(`${process.env.REPLIT_STORAGE_URL}/${filename}`, {
      method: 'PUT',
      body: audioBuffer,
      headers: {
        'Content-Type': 'audio/mpeg'
      }
    });

    if (!storageResponse.ok) {
      throw new Error('Storage upload failed');
    }

    publicUrl = `${process.env.REPLIT_STORAGE_URL}/${filename}`;
  } catch (error) {
    console.error('Storage error:', error);
    return res.status(500).json({ 
      error: 'storage_failed',
      message: 'Failed to save audio file. Please try again. No credits were deducted.'
    });
  }

  // STEP 5: ATOMIC COMMIT - Deduct credit + Save story
  try {
    // Deduct credit
    const newCreditCount = profile.credits - 1;
    await db.set(`profile_${user.id}`, {
      ...profile,
      credits: newCreditCount,
      total_stories: (profile.total_stories || 0) + 1
    });

    // Save story with individual key (prevents race conditions)
    const storyKey = `story_${user.id}_${timestamp}_${storyId}`;
    await db.set(storyKey, {
      id: storyId,
      user_id: user.id,
      hero_name: heroName,
      additional_names: additionalNames,
      generation_type: customPromptText ? 'custom' : 'theme',
      theme: theme,
      custom_prompt_text: customPromptText,
      story_text: storyText,
      audio_url: publicUrl,
      language: language,
      created_at: timestamp
    });

    return res.status(200).json({ 
      success: true, 
      newCreditCount,
      storyId
    });
  } catch (error) {
    console.error('Database commit error:', error);
    return res.status(500).json({ 
      error: 'save_failed',
      message: 'Failed to save story. Contact support if credits were deducted.'
    });
  }
}
```

### 3. API: getStories (pages/api/getStories.ts)

```typescript
import { NextApiRequest, NextApiResponse } from 'next';
import { getUser } from '@replit/replit-auth-api';
import { Client } from '@replit/database';

const db = new Client();

export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  const user = await getUser(req);
  if (!user) return res.status(401).json({ error: 'unauthorized' });

  try {
    // List all story keys for this user
    const prefix = `story_${user.id}_`;
    const keys = await db.list(prefix);
    
    // Fetch all stories
    const stories = await Promise.all(
      keys.map(key => db.get(key))
    );

    // Sort by created_at descending
    stories.sort((a, b) => b.created_at - a.created_at);

    return res.status(200).json({ stories });
  } catch (error) {
    console.error('Get stories error:', error);
    return res.status(500).json({ error: 'failed_to_fetch' });
  }
}
```

### 3. Backend: Stripe Webhook (pages/api/stripe-webhook.ts)

```typescript
import { NextApiRequest, NextApiResponse } from 'next';
import { Client } from '@replit/database';
import Stripe from 'stripe';
import { buffer } from 'micro';

const db = new Client();
const stripe = new Stripe(process.env.STRIPE_SECRET_KEY!, {
  apiVersion: '2023-10-16'
});

// IMPORTANT: Disable body parsing for webhooks
export const config = {
  api: {
    bodyParser: false,
  },
};

export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  if (req.method !== 'POST') return res.status(405).end();

  const buf = await buffer(req);
  const sig = req.headers['stripe-signature']!;

  let event: Stripe.Event;

  try {
    event = stripe.webhooks.constructEvent(
      buf,
      sig,
      process.env.STRIPE_WEBHOOK_SECRET!
    );
  } catch (err: any) {
    console.error('Webhook signature verification failed:', err.message);
    return res.status(400).send(`Webhook Error: ${err.message}`);
  }

  // Handle the checkout.session.completed event
  if (event.type === 'checkout.session.completed') {
    const session = event.data.object as Stripe.Checkout.Session;

    // IMPORTANT: Check for idempotency (Stripe can retry webhooks)
    const paymentIntentId = session.payment_intent as string;
    const alreadyProcessed = await db.get(`payment_${paymentIntentId}`);
    
    if (alreadyProcessed) {
      console.log('Payment already processed:', paymentIntentId);
      return res.status(200).json({ received: true, status: 'already_processed' });
    }

    // Get user ID from client_reference_id
    const userId = session.client_reference_id;
    if (!userId) {
      console.error('No client_reference_id in session');
      return res.status(400).json({ error: 'missing_user_id' });
    }

    // Get the line items to determine which package was purchased
    const lineItems = await stripe.checkout.sessions.listLineItems(session.id, {
      limit: 1
    });

    const priceId = lineItems.data[0]?.price?.id;
    if (!priceId) {
      console.error('No price ID found in line items');
      return res.status(400).json({ error: 'missing_price_id' });
    }

    // Map price ID to credits
    let creditsToAdd = 0;
    if (priceId === process.env.STRIPE_PRICE_5_CREDITS) {
      creditsToAdd = 5;
    } else if (priceId === process.env.STRIPE_PRICE_15_CREDITS) {
      creditsToAdd = 15;
    } else if (priceId === process.env.STRIPE_PRICE_30_CREDITS) {
      creditsToAdd = 30;
    } else {
      console.error('Unknown price ID:', priceId);
      return res.status(400).json({ error: 'unknown_price_id' });
    }

    try {
      // Update user credits
      const profile = await db.get(`profile_${userId}`);
      if (!profile) {
        console.error('Profile not found for user:', userId);
        return res.status(404).json({ error: 'profile_not_found' });
      }

      const newCreditTotal = (profile.credits || 0) + creditsToAdd;
      await db.set(`profile_${userId}`, {
        ...profile,
        credits: newCreditTotal
      });

      // Mark payment as processed (for idempotency)
      await db.set(`payment_${paymentIntentId}`, {
        processed: true,
        user_id: userId,
        credits_added: creditsToAdd,
        timestamp: Date.now()
      });

      console.log(`Added ${creditsToAdd} credits to user ${userId}. New total: ${newCreditTotal}`);

      return res.status(200).json({ 
        received: true, 
        credits_added: creditsToAdd,
        new_total: newCreditTotal
      });
    } catch (error) {
      console.error('Database error in webhook:', error);
      return res.status(500).json({ error: 'database_error' });
    }
  }

  // Return 200 for other event types
  return res.status(200).json({ received: true });
}
```

```typescript
// Cache ElevenLabs voices list
let voicesCache: any = null;
let voicesCacheExpiry = 0;

async function getVoices() {
  if (voicesCache && Date.now() < voicesCacheExpiry) {

  voicesCacheExpiry = Date.now() + 3600000; // 1 hour
  return voicesCache;
}
```

### Database Structure:
```
profile_[user.id] → { email, credits, created_at, total_stories }
story_[user.id]_[timestamp]_[uuid] → { story data }
payment_[payment_intent_id] → { processed, timestamp }
ratelimit_[user.id]_[action] → timestamp
```

### API Routes:
```
POST /api/generateStoryText → Returns story text (free)
POST /api/generateStoryAudio → Returns audio URL (1 credit)
GET  /api/getStories → Returns user's stories
GET  /api/getProfile → Returns user profile
POST /api/stripe-webhook → Handles payments
```
