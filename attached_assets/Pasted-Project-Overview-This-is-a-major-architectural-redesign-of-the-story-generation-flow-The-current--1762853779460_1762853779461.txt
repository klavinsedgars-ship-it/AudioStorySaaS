Project Overview: This is a major architectural redesign of the story generation flow. The current "wait-then-redirect" logic is being removed.

Core Goals:

Eliminate Audio Wait: Implement Audio Streaming from ElevenLabs. The user must hear audio start playing within 5 seconds of clicking "Create Audio."

Stay on Page: The user will stay on the creator.tsx page. The right-hand "Storybook" column will transform into a "loading/playing" state.

Magical Loading: While the audio stream is starting (the new <5s wait), we will show a "magical" loading animation.

Parallel Illustration Generation: We will implement your 5-image carousel idea. The illustrations will be generated in the background while the audio is playing and will magically appear in a carousel one by one.

// server/backgroundJobs.ts
import { storage } from "./storage";
import { LANGUAGE_OPTIONS } from "@shared/schema";
import OpenAI from "openai";
import { Client } from "@replit/object-storage"; 

const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });
const objectStorageClient = new Client({ bucketId: process.env.DEFAULT_OBJECT_STORAGE_BUCKET_ID });

// Your existing ElevenLabs function (modified to take text/lang)
async function generateAudioFromText(text: string, language: string): Promise<ArrayBuffer> {
  const voiceId = LANGUAGE_OPTIONS.find((l) => l.code === language)?.elevenLabsVoiceId || "21m00Tcm4TlvDq8ikWAM";
  const response = await fetch(
    `https://api.elevenlabs.io/v1/text-to-speech/${voiceId}/stream`, // [NEW] Use the /stream endpoint
    {
      method: "POST",
      headers: { "Content-Type": "application/json", "xi-api-key": process.env.ELEVENLABS_API_KEY! },
      body: JSON.stringify({
        text,
        model_id: "eleven_multilingual_v2", // [MODIFIED] Use the stable "v2" model, not "v3"
      }),
    }
  );
  if (!response.ok) throw new Error(`ElevenLabs API error: ${await response.text()}`);
  return response.arrayBuffer(); // This will stream, but for saving we need the whole buffer
}

// Your existing upload function
async function uploadAudioToStorage(audioBuffer: ArrayBuffer, filename: string): Promise<string> {
  const privateDir = process.env.PRIVATE_OBJECT_DIR || ".private";
  const fullPath = `${privateDir}/${filename}`;
  const buffer = Buffer.from(audioBuffer);
  await objectStorageClient.uploadFromBytes(fullPath, buffer);
  return fullPath; // Return the *real path*
}

// === BACKGROUND JOB 1: GENERATE AUDIO ===
export async function generateAudioAndSave(storyId: string, storyText: string, language: string) {
  try {
    await storage.updateStoryStatus(storyId, 'gen_audio');
    
    const audioBuffer = await generateAudioFromText(storyText, language);
    const filename = `story-${storyId}-${Date.now()}.mp3`;
    const audioPath = await uploadAudioToStorage(audioBuffer, filename);
    
    await storage.updateStoryAudio(storyId, audioPath);
    await storage.updateStoryStatus(storyId, 'gen_image'); // Move to next step
    console.log(`[AUDIO] Successfully generated audio for story ${storyId}`);
  } catch (error) {
    console.error(`[AUDIO] Failed to generate audio for story ${storyId}:`, error);
    await storage.updateStoryStatus(storyId, 'failed_audio');
  }
}

// === BACKGROUND JOB 2: GENERATE ILLUSTRATIONS ===
export async function generateIllustrations(storyId: string, storyText: string) {
  try {
    // Wait for audio to be ready
    let story = await storage.getStory(storyId);
    while (story && (story.status === 'pending' || story.status === 'gen_audio')) {
      if (story.status.startsWith('failed')) return;
      await new Promise(resolve => setTimeout(resolve, 3000)); // Poll
      story = await storage.getStory(storyId);
    }

    // 1. Get 5 Scene Prompts from AI
    const promptSummaryResponse = await openai.chat.completions.create({
      model: "gpt-4o", // Use the powerful model for this
      messages: [
        { role: "user", content: `
          You are an AI Art Director. Read this children's story and create 5 distinct visual prompts for DALL-E 3.
          Each prompt must be a "whimsical, magical, children's book illustration" of a key scene.
          Return ONLY a JSON array of 5 strings, like ["prompt 1", "prompt 2", ...].
          
          STORY:
          ${storyText.substring(0, 2000)}...
        `},
      ],
    });
    const prompts: string[] = JSON.parse(summaryResponse.choices[0].message.content || "[]");

    // 2. Generate 5 Images, one by one
    for (const prompt of prompts.slice(0, 5)) { // Ensure only 5
      const imageResponse = await openai.images.generate({
        model: "dall-e-3", prompt, n: 1, size: "1024x1024",
      });
      const imageUrl = imageResponse.data[0].url; // This is a temp URL

      // 3. Download and re-upload to our storage
      const imageFetch = await fetch(imageUrl!);
      const imageBuffer = await imageFetch.arrayBuffer();
      const imageFilename = `image-${storyId}-${Date.now()}.png`;
      const finalImagePath = await uploadAudioToStorage(imageBuffer, imageFilename);

      // 4. Update the DB with the new image URL *immediately*
      await storage.addStoryImageUrl(storyId, finalImagePath);
    }

    // 5. Mark as complete
    await storage.updateStoryStatus(storyId, 'complete');
    console.log(`[IMAGE] Successfully generated all images for story ${storyId}`);

  } catch (error) {
    console.error(`[IMAGE] Failed to generate images for story ${storyId}:`, error);
    await storage.updateStoryStatus(storyId, 'failed_image');
  }
}

// This is how the user gets instant audio.
app.get("/api/stream-audio/:storyId", isAuthenticated, async (req: any, res) => {
  try {
    const userId = req.user.claims.sub;
    const { storyId } = req.params;
    
    const story = await storage.getStory(storyId);
    if (!story || story.userId !== userId) return res.status(403).send("Unauthorized");

    // If file is already generated, stream it from storage
    if (story.audioPath) {
      await streamAudioFile(story.audioPath, req, res, false);
    } else {
      // **Live Stream from ElevenLabs**
      const voiceId = LANGUAGE_OPTIONS.find((l) => l.code === story.language)?.elevenLabsVoiceId || "21m00Tcm4TlvDq8ikWAM";
      const response = await fetch(
        `https://api.elevenlabs.io/v1/text-to-speech/${voiceId}/stream`,
        {
          method: "POST",
          headers: { "Content-Type": "application/json", "xi-api-key": ELEVENLABS_API_KEY! },
          body: JSON.stringify({
            text: story.storyText,
            model_id: "eleven_multilingual_v2",
          }),
        }
      );
      
      if (!response.ok) throw new Error(`ElevenLabs stream error: ${await response.text()}`);

      res.setHeader('Content-Type', 'audio/mpeg');
      response.body.pipe(res);
    }
  } catch (error: any) {
    console.error("Stream audio error:", error);
    res.status(500).send("Failed to stream audio");
  }
});

// [NEW] API Route to poll for illustrations
app.get("/api/story/illustrations/:storyId", isAuthenticated, async (req: any, res) => {
  const userId = req.user.claims.sub;
  const { storyId } = req.params;
  const story = await storage.getStory(storyId);
  
  if (!story || story.userId !== userId) {
    return res.status(404).send("Story not found");
  }
  
  // Return the list of image paths. We'll serve them via the existing audio streamer
  const publicUrls = story.imageUrls.map(path => `/api/audio/${storyId}?imagePath=${encodeURIComponent(path)}`);
  
  res.json({
    status: story.status,
    imageUrls: publicUrls,
  });
});

// [MODIFIED] /api/audio/:storyId to also handle images
app.get("/api/audio/:storyId", isAuthenticated, async (req: any, res) => {
    try {
      const userId = req.user.claims.sub;
      const { storyId } = req.params;

      const story = await storage.getStory(storyId);

        // [NEW] Serve the image
        await streamAudioFile(decodeURIComponent(imagePath as string), req, res, false);
      } else {
        // Serve the audio
        if (!story.audioPath) return res.status(404).send("Audio not available");
        await streamAudioFile(story.audioPath, req, res, false);
      }
    } catch (error: any) {
      res.status(500).send("Failed to serve file");
    }
});

// Hook to poll for illustrations
function useIllustrationPoller(storyId: string | null) {
  const { data, isLoading } = useQuery({
    queryKey: ["/api/story/illustrations", storyId],
    queryFn: async ()_ => {
      const res = await fetch(`/api/story/illustrations/${storyId}`, { credentials: "include" });
      if (!res.ok) throw new Error("Could not fetch illustrations");
      const data = await res.json();
      return data;
    },
    enabled: !!storyId,
    refetchInterval: (data) => (data?.status === 'complete' ? false : 5000), // Poll every 5s until complete
  });
  return { illustrations: data?.imageUrls || [], status: data?.status };
}

  const { data, isLoading } = useQuery<any>({ // Add 'any' type
    queryKey: ["/api/story/illustrations", storyId],
